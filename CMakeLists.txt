cmake_minimum_required(VERSION 3.16)
project(video_gateway C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
option(ASAN "Build with AddressSanitizer" OFF)

find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(LIBCAMERA REQUIRED IMPORTED_TARGET libcamera)

set(SRCS src/main.c src/bufmgr.c src/capture.cpp)
foreach(tgt gateway gateway_baseline)
  add_executable(${tgt} ${SRCS})
  target_include_directories(${tgt} PRIVATE include)
  target_compile_options(${tgt} PRIVATE -Wall -Wextra)
  target_link_libraries(${tgt} PRIVATE PkgConfig::LIBCAMERA Threads::Threads)
  if(ASAN)
    target_compile_options(${tgt} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(${tgt} PRIVATE -fsanitize=address)
  endif()
endforeach()
target_compile_definitions(gateway_baseline PRIVATE BASELINE)

add_executable(m2m_probe tools/m2m_probe.c)
target_compile_options(m2m_probe PRIVATE -Wall -Wextra)
